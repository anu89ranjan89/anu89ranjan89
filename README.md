<br/>

```mermaid
flowchart TB
    U["👩 User"] --> FE["Next.js Frontend<br/>(AWS Amplify)"]
    FE --> GW["API Gateway"]
    GW --> BE["FastAPI Backend<br/>(EC2 + Docker)"]
    BE --> DB[("PostgreSQL<br/>(RDS)")]
    BE --> ML["Explainable ML<br/>PCOS Risk Model"]
    BE --> RAG["RAG Pipeline<br/>LangChain + BGE"]
    RAG --> VS[("FAISS<br/>Vector Index")]
    RAG --> LLM["LLM API"]
    BE --> PDF["PDF Report Generator"]
```
